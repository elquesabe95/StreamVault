// Cargar canales de TVLibre (solo deportes, streams directos) async function loadTVLibreChannels() { try { var res = await fetch('https://apifutboltv.onrender.com/api/channels', { headers: { 'x-authorizer': 'colombia-tv-access' } }); var channels = await res.json(); if (!Array.isArray(channels)) return []; var seen = new Set(); var result = []; for (var i = 0; i < channels.length; i++) { var ch = channels[i]; var name = ch.nombre || ''; var url = ch.url_decoded || ''; if (!name || !url) continue; if (seen.has(name.toLowerCase())) continue; seen.add(name.toLowerCase()); result.push({ name: name, logo: '', category: 'Deportes', country: 'TVLibre', url: url, provider: 'TVLibre' }); } return result; } catch(e) { return []; } }
